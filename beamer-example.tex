\documentclass[aspectratio=169]{beamer}
\usepackage{ascii2tikz}

\title{Draw with ASCII in LaTeX}
\author{}
\date{}

\begin{document}

\begin{frame}[fragile]{ASCII 2 TikZ}
  \centering
  \begin{asciitikz}
    ┌─────────┐   ┌─────────┐   ┌────────┐
    │  Input  │──→│ Process │──→│ Output │
    └─────────┘   └─────────┘   └────────┘
  \end{asciitikz}

  \medskip
  Draw an ASCII diagram in the \texttt{asciitikz} environment, and TikZ code will be
  generated automatically.
\end{frame}

\begin{frame}[fragile]{Options}
  Pass extra TikZ options through the environment's optional argument:
  \centering
  \begin{asciitikz}[scale=4, transform shape]
    ┌─────────────┐
    │ Large Hello │
    └─────────────┘
  \end{asciitikz}
  \medskip
\end{frame}

\end{document}
